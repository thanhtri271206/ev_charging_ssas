/* === CÂU 1: Bảng điểm các mạng sạc trong quý 4 === */
SELECT
    {[Measures].[Avg Utilization],
     [Measures].[OOS Rate],
     [Measures].[Avg Session]} ON 0,
    NON EMPTY [Dim Station].[Network].[Network].Members ON 1
FROM [EV Charging DW]
WHERE ([Dim Date].[Quarter].&[2025]&[4]);


/* === CÂU 2: Sạc nhanh dọc đường cao tốc và sân bay ở từng bang === */
SELECT
    {[Measures].[Avg Utilization], [Measures].[OOS Rate], [Measures].[Avg Wait]} ON 0,
    NON EMPTY [Dim Station].[Geography].[State].Members ON 1
FROM (
    SELECT
        ({[Dim Station].[Location Type].&[Highway Corridor], [Dim Station].[Location Type].&[Airport]} *
         {[Dim Station].[Charger Type].&[DC Fast Charge], [Dim Station].[Charger Type].&[Hyper-Fast]}) ON 0
    FROM [EV Charging DW]
);


/* === CÂU 3a: Xu hướng sử dụng, giá sạc và giá xăng theo Quý === */
SELECT
    {[Measures].[Avg Utilization], [Measures].[Avg Price], [Measures].[Avg Gas]} ON 0,
    [Dim Date].[Calendar].[Quarter].Members ON 1
FROM [EV Charging DW]
WHERE [Dim Station].[Pricing Type].&[per_kwh];


/* === CÂU 3b: Khoan sâu (Drill-down) xu hướng theo từng Tháng === */
SELECT
    {[Measures].[Avg Utilization], [Measures].[Avg Price], [Measures].[Avg Gas]} ON 0,
    [Dim Date].[Calendar].[Month].Members ON 1
FROM [EV Charging DW]
WHERE [Dim Station].[Pricing Type].&[per_kwh];


/* === CÂU 4: Trạm miễn phí (Level 2) có thu hút khách hơn trạm tính phí? === */
SELECT
    {[Measures].[Avg Utilization], [Measures].[Fact Ev Charging Count]} ON 0,
    NON EMPTY
    Crossjoin([Dim Station].[Location Type].[Location Type].Members,
              [Dim Station].[Pricing Type].[Pricing Type].Members) ON 1
FROM [EV Charging DW]
WHERE [Dim Station].[Charger Type].&[Level 2];


/* === CÂU 5: Giờ cao điểm có thật sự đông nhất? (Peak Lift & Peak Premium) === */
SELECT
    {[Measures].[Util Midday], [Measures].[Util Morning Peak], [Measures].[Util Evening Peak],
     [Measures].[Peak Lift Morning], [Measures].[Peak Lift Evening],
     [Measures].[Price Evening], [Measures].[Price Offpeak], [Measures].[Peak Premium %]} ON 0,
    NON EMPTY [Dim Station].[Location Type].[Location Type].Members ON 1
FROM [EV Charging DW];


/* === CÂU 6: Mạng sạc nào chiếm trên 40% lượng sử dụng ở từng bang? === */
WITH
    MEMBER [Measures].[Network Share In State] AS
        IIF(([Measures].[Ports Occupied], [Dim Station].[Network].[All]) = 0, NULL,
            [Measures].[Ports Occupied] / ([Measures].[Ports Occupied], [Dim Station].[Network].[All])),
        FORMAT_STRING = "0.0%"
SELECT
    {[Measures].[Ports Occupied], [Measures].[Network Share In State]} ON 0,
    NON EMPTY
    Filter(
        Crossjoin([Dim Station].[Geography].[State].Members, [Dim Station].[Network].[Network].Members),
        [Measures].[Network Share In State] > 0.4
    ) ON 1
FROM [EV Charging DW];


/* === CÂU 7: Nhu cầu và giá xăng từng thành phố thay đổi MoM (Month-over-Month) === */
WITH
    MEMBER [Measures].[Prev Month Utilization] AS
        ([Dim Date].[Calendar].CurrentMember.PrevMember, [Measures].[Avg Utilization]),
        FORMAT_STRING = "0.0000"
    MEMBER [Measures].[MoM Utilization %] AS
        IIF(IsEmpty([Measures].[Prev Month Utilization]) OR [Measures].[Prev Month Utilization] = 0, NULL,
            ([Measures].[Avg Utilization] - [Measures].[Prev Month Utilization]) / [Measures].[Prev Month Utilization]),
        FORMAT_STRING = "+0.0%;-0.0%;0.0%"
    MEMBER [Measures].[Prev Month Gas] AS
        ([Dim Date].[Calendar].CurrentMember.PrevMember, [Measures].[Avg Gas]),
        FORMAT_STRING = "0.000"
    MEMBER [Measures].[MoM Gas %] AS
        IIF(IsEmpty([Measures].[Prev Month Gas]) OR [Measures].[Prev Month Gas] = 0, NULL,
            ([Measures].[Avg Gas] - [Measures].[Prev Month Gas]) / [Measures].[Prev Month Gas]),
        FORMAT_STRING = "+0.0%;-0.0%;0.0%"
SELECT
    {[Measures].[Avg Utilization], [Measures].[Prev Month Utilization], [Measures].[MoM Utilization %],
     [Measures].[Avg Gas], [Measures].[Prev Month Gas], [Measures].[MoM Gas %]} ON 0,
    NON EMPTY
    Crossjoin([Dim Date].[Calendar].[Month].Members, [Dim Station].[Geography].[City].Members) ON 1
FROM [EV Charging DW];


/* === CÂU 8: Ba trạm kém nhất trong mỗi thành phố (OOS Rate < 10%) === */
SELECT
    {[Measures].[Avg Utilization], [Measures].[OOS Rate], [Measures].[Avg Ports Available]} ON 0,
    NON EMPTY
    Generate(
        [Dim Station].[Geography].[City].Members,
        BottomCount(
            Filter([Dim Station].[Geography].CurrentMember.Children, [Measures].[OOS Rate] < 0.10),
            3,
            [Measures].[Avg Utilization]
        )
    ) ON 1
FROM [EV Charging DW];


/* === CÂU 9: Số giờ-cổng nhu cầu bị mất tích lũy (Cumulative PeriodsToDate) === */
SELECT
    {[Measures].[Lost Port-Hours], [Measures].[Cumulative Lost Port-Hours],
     [Measures].[Cumulative Lost Demand %]} ON 0,
    NON EMPTY
    Crossjoin([Dim Date].[Calendar].[Month].Members, [Dim Station].[Network].[Network].Members) ON 1
FROM [EV Charging DW];


/* === CÂU 10: Ngày bất thường ở 3 thành phố đông nhất trong tháng 11 (MA7 & Z-Score) === */
WITH
    SET [Top3Cities] AS
        TopCount([Dim Station].[Geography].[City].Members, 3,
                 ([Dim Date].[Quarter].&[2025]&[4], [Measures].[Avg Utilization]))
    SET [NovDates] AS
        Descendants([Dim Date].[Calendar].[Month].&[2025]&[4]&[11], [Dim Date].[Calendar].[Date])
    MEMBER [Measures].[MA7 Utilization] AS
        Avg(LastPeriods(7, [Dim Date].[Calendar].CurrentMember), [Measures].[Avg Utilization]),
        FORMAT_STRING = "0.0000"
    MEMBER [Measures].[StdDev7 Utilization] AS
        Stdev(LastPeriods(7, [Dim Date].[Calendar].CurrentMember), [Measures].[Avg Utilization]),
        FORMAT_STRING = "0.0000"
    MEMBER [Measures].[Z Score] AS
        IIF([Measures].[StdDev7 Utilization] = 0 OR IsEmpty([Measures].[StdDev7 Utilization]), NULL,
            ([Measures].[Avg Utilization] - [Measures].[MA7 Utilization]) / [Measures].[StdDev7 Utilization]),
        FORMAT_STRING = "0.00"
SELECT
    {[Measures].[Avg Utilization], [Measures].[MA7 Utilization],
     [Measures].[Z Score], [Measures].[Avg Precipitation]} ON 0,
    NON EMPTY
    Filter(
        Crossjoin([NovDates], [Top3Cities]),
        Abs([Measures].[Z Score]) > 1.5
    ) ON 1
FROM [EV Charging DW];


/* === CÂU 11: Mạng nào chịu áp lực vận hành lớn nhất? (Composite Rank) === */
SELECT
    {[Measures].[Util Evening Peak], [Measures].[Avg Wait],
     [Measures].[Rank Util Evening], [Measures].[Rank Wait],
     [Measures].[Composite Rank], [Measures].[Peak Premium %]} ON 0,
    NON EMPTY
    Order([Dim Station].[Network].[Network].Members, [Measures].[Composite Rank], BASC) ON 1
FROM [EV Charging DW]
WHERE [Dim Station].[Pricing Type].&[per_kwh];


/* === CÂU 12a: Hệ số tương quan kẹt xe, mưa, nhiệt độ theo thành phố === */
WITH
    SET [DaysHot] AS
        Descendants([Dim Date].[Calendar].[Quarter].&[2025]&[3], [Dim Date].[Calendar].[Date])
    SET [DaysCold] AS
        Descendants([Dim Date].[Calendar].[Quarter].&[2025]&[4], [Dim Date].[Calendar].[Date])
    MEMBER [Measures].[r Traffic Hot] AS
        Correlation([DaysHot], [Measures].[Avg Utilization], [Measures].[Avg Traffic]),
        FORMAT_STRING = "0.0000"
    MEMBER [Measures].[r Rain Hot] AS
        Correlation([DaysHot], [Measures].[Avg Utilization], [Measures].[Avg Precipitation]),
        FORMAT_STRING = "0.0000"
    MEMBER [Measures].[r Temp Hot] AS
        Correlation([DaysHot], [Measures].[Avg Utilization], [Measures].[Avg Temperature]),
        FORMAT_STRING = "0.0000"
    MEMBER [Measures].[r Temp Cold] AS
        Correlation([DaysCold], [Measures].[Avg Utilization], [Measures].[Avg Temperature]),
        FORMAT_STRING = "0.0000"
SELECT
    {[Measures].[r Traffic Hot], [Measures].[r Rain Hot], [Measures].[r Temp Hot], [Measures].[r Temp Cold]} ON 0,
    NON EMPTY [Dim Station].[Geography].[City].Members ON 1
FROM [EV Charging DW];


/* === CÂU 12b: Chênh lệch mức sử dụng ngày thời tiết cực đoan (Extreme Lift %) === */
WITH
    MEMBER [Measures].[Avg Util Normal Weather] AS
        ([Fact Ev Charging].[Weather Condition].&[clear], [Measures].[Avg Utilization]),
        FORMAT_STRING = "0.0000"
    MEMBER [Measures].[Avg Util Extreme] AS
        Aggregate(
            {[Fact Ev Charging].[Weather Condition].&[extreme_heat],
             [Fact Ev Charging].[Weather Condition].&[freezing]},
            [Measures].[Avg Utilization]
        ),
        FORMAT_STRING = "0.0000"
    MEMBER [Measures].[Extreme Lift %] AS
        IIF([Measures].[Avg Util Normal Weather] = 0, NULL,
            ([Measures].[Avg Util Extreme] - [Measures].[Avg Util Normal Weather]) /
            [Measures].[Avg Util Normal Weather]),
        FORMAT_STRING = "+0.0%;-0.0%;0.0%"
SELECT
    {[Measures].[Avg Utilization], [Measures].[Avg Util Normal Weather],
     [Measures].[Avg Util Extreme], [Measures].[Extreme Lift %]} ON 0,
    [Dim Station].[Geography].[City].Members ON 1
FROM [EV Charging DW];


/* === CÂU 13: Tác động của Sự kiện và Mưa lớn theo loại địa điểm === */
WITH
    SET [Events] AS
        Except([Fact Ev Charging].[Local Event].[Local Event].Members,
               {[Fact Ev Charging].[Local Event].&[none]})
    MEMBER [Fact Ev Charging].[Local Event].[Has Event] AS
        Aggregate([Events])
    MEMBER [Measures].[Avg Util Normal] AS
        ([Fact Ev Charging].[Local Event].&[none], [Fact Ev Charging].[Weather Condition].&[clear],
         [Measures].[Avg Utilization]), FORMAT_STRING = "0.0000"
    MEMBER [Measures].[Avg Util Event Only] AS
        ([Fact Ev Charging].[Local Event].[Has Event], [Fact Ev Charging].[Weather Condition].&[clear],
         [Measures].[Avg Utilization]), FORMAT_STRING = "0.0000"
    MEMBER [Measures].[Avg Util Rain Only] AS
        ([Fact Ev Charging].[Local Event].&[none], [Fact Ev Charging].[Weather Condition].&[heavy_rain],
         [Measures].[Avg Utilization]), FORMAT_STRING = "0.0000"
    MEMBER [Measures].[Avg Util Event And Rain] AS
        ([Fact Ev Charging].[Local Event].[Has Event], [Fact Ev Charging].[Weather Condition].&[heavy_rain],
         [Measures].[Avg Utilization]), FORMAT_STRING = "0.0000"
    MEMBER [Measures].[Rain Impact No Event] AS
        [Measures].[Avg Util Rain Only] - [Measures].[Avg Util Normal], FORMAT_STRING = "+0.0000;-0.0000;0.0000"
    MEMBER [Measures].[Rain Impact With Event] AS
        [Measures].[Avg Util Event And Rain] - [Measures].[Avg Util Event Only], FORMAT_STRING = "+0.0000;-0.0000;0.0000"
SELECT
    {[Measures].[Avg Util Normal], [Measures].[Avg Util Event Only],
     [Measures].[Avg Util Rain Only], [Measures].[Avg Util Event And Rain],
     [Measures].[Rain Impact No Event], [Measures].[Rain Impact With Event],
     [Measures].[Avg Wait], [Measures].[Avg Session]} ON 0,
    NON EMPTY [Dim Station].[Location Type].[Location Type].Members ON 1
FROM [EV Charging DW];


/* === CÂU 14: Phân khúc (TP x Loại địa điểm) tăng nhanh nhưng độ tin cậy kém (QoQ & OOS > 10%) === */
SELECT
    {[Measures].[Avg Utilization], [Measures].[Prev Quarter Utilization], [Measures].[QoQ Utilization %],
     [Measures].[OOS Rate], [Measures].[Prev Quarter OOS], [Measures].[Fact Ev Charging Count]} ON 0,
    NON EMPTY
    Filter(
        Crossjoin([Dim Station].[Geography].[City].Members,
                  [Dim Station].[Location Type].[Location Type].Members),
        NOT IsEmpty([Measures].[QoQ Utilization %])
        AND [Measures].[OOS Rate] > 0.10
        AND [Measures].[Fact Ev Charging Count] > 0
    ) ON 1
FROM [EV Charging DW]
WHERE [Dim Date].[Quarter].&[2025]&[4];


/* === CÂU 15a: Thành phố quá tải nhất vào khung tối tháng 12 === */
WITH 
    MEMBER [Measures].[Surge Exposure %] AS
        IIF([Measures].[Fact Ev Charging Count] = 0, NULL,
            ([Fact Ev Charging].[Utilization Level].&[Near Capacity], [Measures].[Fact Ev Charging Count]) / 
            [Measures].[Fact Ev Charging Count]),
        FORMAT_STRING = "0.00%"
SELECT [Measures].[Surge Exposure %] ON 0,
    Order([Dim Station].[Geography].[City].Members, [Measures].[Surge Exposure %], BDESC) ON 1
FROM [EV Charging DW]
WHERE ([Dim Date].[Month].&[2025]&[4]&[12], [Dim Time].[Price Period].&[Evening Peak]);
